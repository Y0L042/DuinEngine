46 // ---- term methods ----
47 // Each type becomes its own AND-ed filter term, so With<A, B>() is
48 // equivalent to chaining .With<A>().With<B>(). We expand the pack into one
49 // with<T>() call per type rather than forwarding the whole pack: flecs'
50 // with<First, Second>() is the relationship-pair overload, which would build
51 // a single (A, B) pair term instead of two separate terms.
52 template <typename... Filters>
53 SystemBuilder<Components...> &With()
54 {
55 (flecsSystemBuilder->template with<Filters>(), ...);
56 return *this;
57 }
58
59 // Each type becomes its own negated term (excludes entities having any of
60 // them), matching .Without<A>().Without<B>(). See With() above for why the
61 // pack is expanded per-type rather than forwarded to without<First, Second>().
62 template <typename... Filters>
63 SystemBuilder<Components...> &Without()
64 {
65 (flecsSystemBuilder->template without<Filters>(), ...);

143 template <typename Func>
144 System Each(Func &&func)
145 {
146 // NOTE: a system callback is stored and invoked later (every progress),
147 // so the functor must be captured BY VALUE. This differs from
148 // Query::Each, which captures by reference because the query callback
149 // runs synchronously and immediately. The flecs entity passed to the
150 // callback already carries its own world.
151 flecs::system sys = flecsSystemBuilder->each(
152 [func = std::forward<Func>(func)](
153 flecs::entity flecsEntity,
154 std::conditional_t<std::is_pointer_v<Components>, Components, Components &>... comps) {
155 duin::Entity duinEntity(flecsEntity);
156 func(duinEntity, comps...);
157 });
158 return System(std::move(sys));
159 }
160
169 template <typename Func>
170 System Run(Func &&func)
171 {
172 flecs::system sys = flecsSystemBuilder->run([func = std::forward<Func>(func)](flecs::iter &flecsIter) {
173 duin::Iter duinIter(flecsIter);
174 func(duinIter);
175 });
176 return System(std::move(sys));
177 }
