182 {
183 flecsWorld.children([&](flecs::entity child) {
184 if (filterBuiltins)
185 {
186 // Filter out known FLECS internal entity categories:
187 // flecs::Component (struct type) component/tag type descriptor entities
188 // flecs::Observer (entity_t tag) internal observer entities
189 // flecs::Module (entity_t tag) module scope entities

193 return;
194 if (child.has(flecs::Module))
195 return;

199 e.SetFlecsEntity(child);
200 f(e);
201 });
202 }
