165 template <typename Func>
166 void IterateChildren(Func &&f, bool filterBuiltins = true) const
167 {
168 flecsWorld.children([&](flecs::entity child) {
169 if (filterBuiltins)
170 {
171 // Filter out known FLECS internal entity categories:
172 // flecs::Component (struct type) component/tag type descriptor entities
173 // flecs::Observer (entity_t tag) internal observer entities
174 // flecs::Module (entity_t tag) module scope entities
175 if (child.has<flecs::Component>())
176 return;
177 if (child.has(flecs::Observer))
178 return;
179 if (child.has(flecs::Module))
180 return;
181 }
182 // child already carries its own world.
183 duin::Entity e(child);
184 f(e);
185 });
186 }
