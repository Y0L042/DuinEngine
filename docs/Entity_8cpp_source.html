152duin::Entity &duin::Entity::SetName(const std::string &name)
153{
154 if (!RequireValid())
155 {
156 return *this;
157 }
158 Entity e;
159 if (Parent().IsValid())
160 {
161 e = Parent().Lookup(name);
162 }
163 else
164 {
165 e = world->Lookup(name);
166 }
167
168 DN_CORE_ASSERT(!e.IsValid(), "No duplicate names allowed!");
169 if (!e.IsValid())
170 {
171 flecsEntity.set_name(name.c_str());
172 }
173 else
174 {
175 DN_CORE_WARN(
176 "Entity {} with name <{}> already exist when trying to set name for entity {}.",
177 e.flecsEntity.raw_id(),
178 name,
179 flecsEntity.raw_id());
180 }
181
182 return *this;
183}
