46// DN asserts route through duin::ReportAssert (defined once in Duin.lib) so the
47// macro expands identically in every TU. The old CHECK/WARN expansion baked the
48// fail-vs-log choice into header-inline code, which leaked via ODR from Duin.lib
49// and failed unrelated DuinTests cases. ReportAssert logs but never fails a test
50// (use doctest CHECK/REQUIRE for real expectations).
51// DN_ALLOW_ASSERT_FAILS is now a no-op: log-don't-fail is the default.
52
53// DN asserts behave like standard assert: active in debug builds, compiled out
54// entirely in non-debug (release) builds. Use them for invariants/bugs only;
55// for runtime failures that can legitimately happen, return std::expected.
56// msg is an fmt-style format string; trailing args fill its {} placeholders:
57// DN_CORE_ASSERT(p != nullptr, "ptr {} was null at index {}", name, i);
58#ifdef DN_DEBUG
59
60#define DN_CORE_ASSERT(condition, ...) \
61 ((condition) ? (void)0 : ::duin::ReportAssertFmt(#condition, __FILE__, __LINE__, __VA_ARGS__))
62#define DN_CORE_ASSERT_NOMSG(condition) \
63 ((condition) ? (void)0 : ::duin::ReportAssert(#condition, nullptr, __FILE__, __LINE__))
64#define DN_ASSERT(condition, ...) DN_CORE_ASSERT(condition, __VA_ARGS__)
65#define DN_ASSERT_NOMSG(condition) DN_CORE_ASSERT_NOMSG(condition)
66
67#else
